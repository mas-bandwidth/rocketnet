cmake_minimum_required(VERSION 3.20)
project(rocketnet_consumer LANGUAGES C CXX)
if(ROCKETNET_SOURCE_PATH)
    add_subdirectory("${ROCKETNET_SOURCE_PATH}" rocketnet-build)
    foreach(unwanted test client server keygen soak soak_sim)
        if(TARGET ${unwanted})
            message(FATAL_ERROR "Embedded SDK unexpectedly added ${unwanted}")
        endif()
    endforeach()
    get_target_property(static_type rocketnet TYPE)
    if(NOT static_type STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "Parent BUILD_SHARED_LIBS changed the static target")
    endif()
else()
    find_package(RocketNet 1.0 CONFIG REQUIRED)
    # the package's full release label must be the installed
    # header's ROCKETNET_VERSION_FULL; main.c compares the two
    if(NOT RocketNet_VERSION_STRING)
        message(FATAL_ERROR "RocketNet package carries no RocketNet_VERSION_STRING")
    endif()
    set(ROCKETNET_PACKAGE_VERSION_STRING "${RocketNet_VERSION_STRING}")
endif()
enable_testing()
foreach(linkage rocketnet rocketnet_shared)
    foreach(language c cpp)
        set(consumer "consumer_${linkage}_${language}")
        add_executable(${consumer} main.${language})
        target_link_libraries(${consumer} PRIVATE RocketNet::${linkage})
        # The installed headers reach consumers as ordinary includes here, so
        # their warnings are not silenced as system headers.
        set_target_properties(${consumer} PROPERTIES NO_SYSTEM_FROM_IMPORTED ON)
        if(MSVC)
            target_compile_options(${consumer} PRIVATE /W4 /WX)
        else()
            target_compile_options(${consumer} PRIVATE -Wall -Wextra -Werror)
        endif()
        if(linkage STREQUAL "rocketnet_shared")
            target_compile_definitions(${consumer} PRIVATE EXPECT_SHARED=1)
        endif()
        if(ROCKETNET_PACKAGE_VERSION_STRING)
            target_compile_definitions(${consumer} PRIVATE
                "EXPECT_VERSION_FULL=\"${ROCKETNET_PACKAGE_VERSION_STRING}\"")
        endif()
        add_test(NAME ${consumer} COMMAND ${consumer})
    endforeach()
endforeach()
